+     contact.name = contact.name.trim();

+   return contact;

+ // Dedup runs on the normalized record so records that only differ by
+ // case/whitespace (e.g. "ADA@Example.com" vs "ada@example.com") collapse
+ // into one, keeping the first occurrence.
+ const seen = new Set();
+ /** @type {Contact[]} */
+ const deduped = [];
+ for (const contact of normalized) {
+   const key = JSON.stringify(
+     Object.keys(contact)
+       .sort()
+       .map((k) => [k, contact[k]]),
+   );
+   if (seen.has(key)) continue;
+   seen.add(key);
+   deduped.push(contact);
+ }
+ 
+ return deduped.map((json) => ({ json }));
