+ /** @typedef {{ name?: unknown, email?: unknown, [key: string]: unknown }} ContactRecord */

+ const items = $input.all();

+ /** @type {Map<string, ContactRecord>} */
+ const seen = new Map();
+ 
+ for (const item of items) {
+   /** @type {ContactRecord} */
+   const record = { ...item.json };
+ 
+   if (typeof record.name === "string") {
+     record.name = record.name.trim();

+   if (typeof record.email === "string") {
+     record.email = record.email.trim().toLowerCase();
+   }
+ 
+   const key = JSON.stringify(record, Object.keys(record).sort());
+   if (!seen.has(key)) {
+     seen.set(key, record);
+   }
+ }
+
