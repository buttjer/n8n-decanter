+ // @ts-check

+ // Normalize — cleans a pasted array of contact records.
+ //
+ //   1. Normalise: emails lowercased and trimmed, names trimmed.
+ //   2. Drop duplicates, comparing records *after* normalisation, so rows that
+ //      differ only in casing/padding collapse into one. First occurrence wins,
+ //      and the original input order is preserved.
+ //
+ // Normalising before de-duplicating is what makes the two "Ada" rows in the
+ // sample input a duplicate pair; raw-exact duplicates are still caught, since
+ // identical records stay identical once normalised.
+ //
+ // Mode: runOnceForAllItems.
+ 
+ /** @typedef {Record<string, any>} Contact */
+ 
+ /**
+  * A stable string key for a whole record. Object keys are sorted so that two
+  * records with the same fields in a different order still count as duplicates.
+  * @param {unknown} value
+  * @returns {string}
+  */
+ function stableKey(value) {
+   if (Array.isArray(value)) {
+     return `[${value.map(stableKey).join(",")}]`;
+   }
+   if (value !== null && typeof value === "object") {
+     const record = /** @type {Contact} */ (value);
+     const fields = Object.keys(record)
+       .sort()
+       .map((key) => `${JSON.stringify(key)}:${stableKey(record[key])}`);
+     return `{${fields.join(",")}}`;
+   }
+   // undefined has no JSON representation; keep it distinct from null.
+   return value === undefined ? "undefined" : JSON.stringify(value);
+ }
+ 
+ /**
+  * @param {Contact} record
+  * @returns {Contact}
+  */
+ function normalizeContact(record) {
+   const normalized = { ...record };
+   if (typeof normalized.email === "string") {
+     normalized.email = normalized.email.trim().toLowerCase();
+   }
+   if (typeof normalized.name === "string") {
+     normalized.name = normalized.name.trim();
+   }
+   return normalized;
+ }
+ 
+ // Collect the incoming records. Pasting a JSON array normally gives one item
+ // per record; the guard also covers the whole array arriving as a single item.
+ /** @type {Contact[]} */
+ const records = [];
+ for (const item of $input.all()) {
+   if (Array.isArray(item.json)) {
+     records.push(...item.json);
+   } else {
+     records.push(item.json);
+   }
+ }
+ 
+ /** @type {Set<string>} */
+ const seen = new Set();
+ /** @type {Contact[]} */
+ const unique = [];
+ 
+ for (const record of records) {
+   const normalized = normalizeContact(record);
+   const key = stableKey(normalized);
+   if (seen.has(key)) continue;
+   seen.add(key);
+   unique.push(normalized);
+ }
+ 
+ return unique.map((json) => ({ json }));

+// Normalize — cleans a pasted array of contact records.
+//
+//   1. Normalise: emails lowercased and trimmed, names trimmed.
+//   2. Drop duplicates, comparing records *after* normalisation, so rows that
+//      differ only in casing/padding collapse into one. First occurrence wins,
+//      and the original input order is preserved.
+//
+// Normalising before de-duplicating is what makes the two "Ada" rows in the
+// sample input a duplicate pair; raw-exact duplicates are still caught, since
+// identical records stay identical once normalised.
+//
+// Mode: runOnceForAllItems.
+
+/** @typedef {Record<string, any>} Contact */
+
+/**
+ * A stable string key for a whole record. Object keys are sorted so that two
+ * records with the same fields in a different order still count as duplicates.
+ * @param {unknown} value
+ * @returns {string}
+ */
+function stableKey(value) {
+  if (Array.isArray(value)) {
+    return `[${value.map(stableKey).join(",")}]`;
+  }
+  if (value !== null && typeof value === "object") {
+    const record = /** @type {Contact} */ (value);
+    const fields = Object.keys(record)
+      .sort()
+      .map((key) => `${JSON.stringify(key)}:${stableKey(record[key])}`);
+    return `{${fields.join(",")}}`;
+  }
+  // undefined has no JSON representation; keep it distinct from null.
+  return value === undefined ? "undefined" : JSON.stringify(value);
+}
+
+/**
+ * @param {Contact} record
+ * @returns {Contact}
+ */
+function normalizeContact(record) {
+  const normalized = { ...record };
+  if (typeof normalized.email === "string") {
+    normalized.email = normalized.email.trim().toLowerCase();
+  }
+  if (typeof normalized.name === "string") {
+    normalized.name = normalized.name.trim();
+  }
+  return normalized;
+}
+
+// Collect the incoming records. Pasting a JSON array normally gives one item
+// per record; the guard also covers the whole array arriving as a single item.
+/** @type {Contact[]} */
+const records = [];
+for (const item of $input.all()) {
+  if (Array.isArray(item.json)) {
+    records.push(...item.json);
+  } else {
+    records.push(item.json);
+  }
+}
+
+/** @type {Set<string>} */
+const seen = new Set();
+/** @type {Contact[]} */
+const unique = [];
+
+for (const record of records) {
+  const normalized = normalizeContact(record);
+  const key = stableKey(normalized);
+  if (seen.has(key)) continue;
+  seen.add(key);
+  unique.push(normalized);
+}
