+/** Matches a field value that looks like an email address. */
+const EMAIL_PATTERN = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
+
+/**
+ * Recursively sorts object keys so two records with the same data in a
+ * different key order still produce the same fingerprint.
+ * @param {unknown} value
+ * @returns {unknown}
+ */
+function canonicalize(value) {
+  if (Array.isArray(value)) {
+    return value.map(canonicalize);
+  }
+  if (value && typeof value === "object") {
+    /** @type {Record<string, unknown>} */
+    const sorted = {};
+    for (const key of Object.keys(value).sort()) {
+      sorted[key] = canonicalize(/** @type {Record<string, unknown>} */ (value)[key]);
+    }
+    return sorted;
+  }
+  return value;
+}
