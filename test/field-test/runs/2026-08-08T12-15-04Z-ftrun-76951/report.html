+ /**
+  * @typedef {Record<string, unknown>} ContactRecord
+  */

+ /**
+  * Deterministic string form of a value, independent of object key order,
+  * so two records that differ only in key order still compare equal.
+  * @param {unknown} value
+  * @returns {string}
+  */
+ function canonicalize(value) {
+   if (Array.isArray(value)) {
+     return `[${value.map(canonicalize).join(",")}]`;
+   }
+   if (value !== null && typeof value === "object") {
+     const record = /** @type {ContactRecord} */ (value);
+     const keys = Object.keys(record).sort();
+     return `{${keys
+       .map((key) => `${JSON.stringify(key)}:${canonicalize(record[key])}`)
+       .join(",")}}`;
+   }
+   return JSON.stringify(value);
+ }
+ 
+ /**
+  * @param {ContactRecord} record
+  * @returns {ContactRecord}
+  */
+ function normalizeRecord(record) {
+   const normalized = { ...record };
+   if (typeof normalized.email === "string") {
+     normalized.email = normalized.email.trim().toLowerCase();
+   }
+   if (typeof normalized.name === "string") {
+     normalized.name = normalized.name.trim();
+   }
+   return normalized;
+ }

+/**
+ * Deterministic string form of a value, independent of object key order,
+ * so two records that differ only in key order still compare equal.
+ * @param {unknown} value
+ * @returns {string}
+ */
+function canonicalize(value) {
+  if (Array.isArray(value)) {
+    return `[${value.map(canonicalize).join(",")}]`;
+  }
+  if (value !== null && typeof value === "object") {
+    const record = /** @type {ContactRecord} */ (value);
+    const keys = Object.keys(record).sort();
+    return `{${keys
+      .map((key) => `${JSON.stringify(key)}:${canonicalize(record[key])}`)
+      .join(",")}}`;
+  }
+  return JSON.stringify(value);
+}
+
+/**
+ * @param {ContactRecord} record
+ * @returns {ContactRecord}
+ */
+function normalizeRecord(record) {
+  const normalized = { ...record };
+  if (typeof normalized.email === "string") {
+    normalized.email = normalized.email.trim().toLowerCase();
+  }
+  if (typeof normalized.name === "string") {
+    normalized.name = normalized.name.trim();
+  }
